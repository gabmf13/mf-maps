      rows[0].split(",");

      .slice(1)
      .filter((row) => row.trim() !== "")
      .map((row) => {

        const cols = row.split(",");

        const obj = {};

        headers.forEach((header, index) => {

            cols[index]?.trim() || "";

        return obj;

function openWaze(lat, lng) {

  if (!lat || !lng) return;

  window.location.href =
    `waze://?ll=${lat},${lng}&navigate=yes`;

}

/* NAVIGATION APP */
function changeTab(newTab) {

  window.history.pushState(
    { tab: newTab },
    ""
  );

  setTab(newTab);

}

/* SWIPE / RETOUR */
useEffect(() => {

  window.history.replaceState(
    { tab: "list" },
    ""
  );

  const handlePopState = (event) => {

    if (event.state?.tab) {

      changeTab(event.state.tab);

    } else {

      changeTab("list");

    }

  };

  window.addEventListener(
    "popstate",
    handlePopState
  );

  return () => {

    window.removeEventListener(
      "popstate",
      handlePopState
    );

  };

}, []);

    changeTab={changeTab}
  />
)}

{tab === "contact-radar" && (
  <ContactPage
    title="Contacts Radars"
    data={contactPages.radar}
    changeTab={changeTab}
  />
)}

{tab === "contact-autre" && (
  <ContactPage
    title="Contacts Autres"
    data={contactPages.autre}
    changeTab={changeTab}
  />
)}




</div>

        {/* DETAIL */}
        {selected && (

          <div
  className="
  fixed
  inset-0
  z-[100]
  bg-black/40
  backdrop-blur-sm
  flex
  justify-center
  items-end
  px-3
  "
>
            <div
  className="
  w-full
  max-w-md
  mx-auto
  max-h-[88vh]
  bg-[#0f172a]
  rounded-t-[34px]
  overflow-y-auto
  shadow-[0_-20px_60px_rgba(0,0,0,0.45)]
  animate-[slideUp_.35s_ease-out]
  "
>

            {/* TOP */}
            <div className="sticky top-0 bg-[#0f172a] z-20 px-5 pt-5 pb-5 border-b border-slate-800">

              <button
                onClick={() => setSelected(null)}
                className="absolute top-5 right-5 w-10 h-10 rounded-full bg-slate-800 text-white flex items-center justify-center"
              >
                ✕
              </button>



              <h2 className="text-xl font-bold text-white break-words pr-12">
                {selected.station}
              </h2>

              
              <div className="flex items-center gap-3 mt-3 flex-wrap">
                <p className="text-slate-400">
                  INSEE : {selected.insee}
                </p>

                <div
                  className={`
                    inline-flex items-center px-3 py-1.5 rounded-full text-[11px] font-semibold
                    ${badgeColors[selected.type] || "bg-slate-700 text-slate-300"}
                  `}
                >
                  {selected.type}
                </div>
              </div>


            </div>

            {/* INFOS */}
            <div className="px-4 py-5">

              <div className="space-y-4">


                <button
                  onClick={() =>
                    openWaze(
                      selected.latitude,
                      selected.longitude
                    )
                  }
                  className="w-full bg-[#1677ff] text-white rounded-[22px] py-4 font-semibold flex items-center justify-center gap-2"
                >
                  <Navigation size={20} />
                  Ouvrir dans Waze
                </button>

                <div className="bg-slate-700/80 rounded-[24px] p-5">

                  <div className="flex items-center justify-between mb-2">
                  <p className="text-slate-300 text-sm">
                    Coordonnées
                  </p>

                  <button
                    onClick={() =>
                      navigator.clipboard.writeText(
                        `${selected.latitude}, ${selected.longitude}`
                      )
                    }
                    className="text-slate-300 hover:text-white"
                  >
                    <Copy size={18} />
                  </button>
                </div>

                  <p>
                    {selected.latitude}, {selected.longitude}
                  </p>

                </div>

                {(selected.nom1 || selected.nom2) && (
                <div className="bg-slate-700/80 rounded-[24px] p-5">

                  <p className="text-slate-300 mb-3 text-sm">
                    Contacts
                  </p>

                  <div className="space-y-3">

                    {selected.nom1 && (
                      <a
                        href={`tel:${selected.numero1}`}
                        className="flex items-center justify-between"
                      >
                        <div>
                          <p className="font-semibold text-white">
                            {selected.nom1}
                          </p>
                          <p className="text-sky-400 text-sm">
                            {selected.numero1}
                          </p>
                        </div>
                        <Phone size={20} className="text-sky-400" />
                      </a>
                    )}

                    {selected.nom2 && (
                      <a
                        href={`tel:${selected.numero2}`}
                        className="flex items-center justify-between"
                      >
                        <div>
                          <p className="font-semibold text-white">
                            {selected.nom2}
                          </p>
                          <p className="text-sky-400 text-sm">
                            {selected.numero2}
                          </p>
                        </div>
                        <Phone size={20} className="text-sky-400" />
                      </a>
                    )}

                  </div>

                </div>
                )}

                <div className="bg-slate-700/80 rounded-[24px] p-5">

                  <div className="flex items-center justify-between mb-2">
                  <p className="text-slate-300 text-sm">
                    Identifiant SIM
                  </p>

                  <button
                    onClick={() =>
                      navigator.clipboard.writeText(
                        selected.sim || ""
                      )
                    }
                    className="text-slate-300 hover:text-white"
                  >
                    <Copy size={18} />
                  </button>
                </div>

                  <p>
                    {selected.sim || "Non renseigné"}
                  </p>

                </div>

                <div className="bg-slate-700/80 rounded-[24px] p-5">

                  <div className="flex items-center justify-between mb-2">
                  <p className="text-slate-300 text-sm">
                    Mot de passe SIM
                  </p>

                  <button
                    onClick={() =>
                      navigator.clipboard.writeText(
                        selected.password || ""
                      )
                    }
                    className="text-slate-300 hover:text-white"
                  >
                    <Copy size={18} />
                  </button>
                </div>

                  <p>
                    {selected.password || "Non renseigné"}
                  </p>

                </div>
{selected.formulaire && (
    <div className="bg-slate-700/80 rounded-[24px] p-5">
      <p className="text-slate-300 mb-3 text-sm">
        Formulaire préventif
      </p>

      <button
        onClick={() =>
          window.location.href = selected.formulaire
        }
        className="
          w-full
          bg-[#1677ff]
          text-white
          rounded-[18px]
          py-4
          font-semibold
          flex
          items-center
          justify-center
          gap-2
        "
      >
        <FileText size={20} />
        Ouvrir le formulaire
      </button>
    </div>
)}
                <div className="bg-slate-700/80 rounded-[24px] p-5">

                  <p className="text-slate-300 mb-3 text-sm">
                    Notes
                  </p>

                  <textarea
                    defaultValue={selected.notes}
                    readOnly className="w-full min-h-[140px] rounded-[18px] bg-slate-700 border border-slate-600 p-4 resize-none"
                  />

                </div>

              </div>

            </div>

            </div>
          </div>

        )}

        {/* NAVBAR */}
        {!selected && (
        <div
  className="
  fixed
  bottom-5
  left-1/2
  -translate-x-1/2
  w-[calc(100%-24px)]
  max-w-[calc(28rem-24px)]
  bg-slate-800/90
  backdrop-blur-xl
  rounded-[28px]
  border border-slate-700 shadow-[0_15px_40px_rgba(0,0,0,0.15)]
  py-3
  px-2
  flex
  justify-around
  z-50
"
>

  <button
    onClick={() => changeTab("list")}
    className={`flex flex-col items-center gap-1 ${
      tab === "list"
        ? "text-[#1677ff]"
        : "text-slate-300"
    }`}
  >
    <List size={22} />
    <span className="text-xs font-medium">
      Liste
    </span>
  </button>

  <button
    onClick={() => changeTab("map")}
    className={`flex flex-col items-center gap-1 ${
      tab === "map"
        ? "text-[#1677ff]"
        : "text-slate-300"
    }`}
  >
    <Map size={22} />
    <span className="text-xs font-medium">
      Carte
    </span>
  </button>

  <button
    onClick={() => changeTab("contact")}
    className={`flex flex-col items-center gap-1 ${
      tab.includes("contact")
        ? "text-[#1677ff]"
        : "text-slate-300"
    }`}
  >
    <Users size={22} />
    <span className="text-xs font-medium">
      Contacts
    </span>
  </button>

</div>
)}
</div>

</main>
  );
}
